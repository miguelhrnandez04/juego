-- Ranking de "Camino a Dueño" en Supabase -- Pegar todo en Supabase → SQL Editor → New query → Run. -- El juego (con la clave pública "anon") solo puede SUMAR partidas y nombres y LEER el ranking. -- No puede modificar ni borrar nada, y no ve el identificador interno de cada jugador. -- 1) Partidas que llegaron a la entrega de llaves create table if not exists public.scores ( id bigint generated always as identity primary key, player_id uuid not null, name text check (name is null or char_length(name) between 1 and 20), branch text not null check (char_length(branch) between 1 and 40), project text check (project is null or char_length(project) <= 60), role text check (role in ('promotor', 'promotora')), time_s real not null check (time_s between 60 and 3600), -- descarta tiempos imposibles coins int not null default 0 check (coins between 0 and 5000), created_at timestamptz not null default now() ); -- 2) Nombres que elige cada jugador (se usa siempre el último) create table if not exists public.nombres ( id bigint generated always as identity primary key, player_id uuid not null, name text not null check (char_length(name) between 1 and 20), created_at timestamptz not null default now() ); create index if not exists scores_player_idx on public.scores (player_id); create index if not exists nombres_player_idx on public.nombres (player_id, created_at desc); -- 3) Seguridad: solo agregar filas, nunca leerlas ni cambiarlas directamente alter table public.scores enable row level security; alter table public.nombres enable row level security; drop policy if exists "sumar partida" on public.scores; drop policy if exists "poner nombre" on public.nombres; create policy "sumar partida" on public.scores for insert to anon with check (true); create policy "poner nombre" on public.nombres for insert to anon with check (true); revoke all on public.scores, public.nombres from anon, authenticated; grant insert on public.scores, public.nombres to anon; -- 4) Lo que el juego puede leer: el ranking ya armado create or replace view public.ranking_jugadores as with s as ( select player_id, min(time_s) as best_time, max(coins) as best_coins, count(*) as wins, (array_agg(branch order by created_at desc))[1] as branch from public.scores group by player_id ), n as ( select distinct on (player_id) player_id, name from ( select player_id, name, created_at from public.nombres union all select player_id, name, created_at from public.scores where name is not null ) x order by player_id, created_at desc ) select encode(sha256(convert_to(s.player_id::text, 'UTF8')), 'hex') as pid, -- código público, no el identificador real n.name, s.branch, s.best_time, s.best_coins, s.wins from s left join n using (player_id); create or replace view public.ranking_sucursales as select branch, count(*)::int as wins from public.scores group by branch; revoke all on public.ranking_jugadores, public.ranking_sucursales from anon, authenticated; grant select on public.ranking_jugadores, public.ranking_sucursales to anon;
